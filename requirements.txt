# ============================================================
# AI Learning Assistant — Dependency Reference
# ============================================================
# This project uses Node.js (not Python), so this file serves
# as a human-readable reference for all dependencies.
# Run the install commands listed under each section.
# ============================================================

# ── BACKEND (Node.js / Express) ──────────────────────────────
# Runtime: Node.js >= 18.x  |  npm >= 9.x

express@^4.18.2          # HTTP server & routing
cors@^2.8.5              # Cross-origin requests (frontend ↔ backend)
dotenv@^16.4.5           # Load .env variables
axios@^1.7.2             # HTTP client to call Groq API
groq-sdk@^0.3.3          # Official Groq Node SDK (optional but recommended)
express-validator@^7.1.0 # Input validation middleware
morgan@^1.10.0           # HTTP request logger

# Install backend deps:
#   cd backend && npm install

# ── FRONTEND (React + Vite) ───────────────────────────────────
# Runtime: Node.js >= 18.x

react@^18.3.1            # UI library
react-dom@^18.3.1        # DOM renderer
vite@^5.3.1              # Build tool / dev server
@vitejs/plugin-react      # Vite React plugin
axios@^1.7.2             # API calls from frontend
framer-motion@^11.3.2    # Animations & transitions
lucide-react@^0.400.0    # Icon library

# Install frontend deps:
#   cd frontend && npm install

# ── ENVIRONMENT VARIABLES NEEDED ─────────────────────────────
# Create backend/.env with:
#   GROQ_API_KEY=your_groq_api_key_here
#   PORT=5000

# ── NODE VERSION ─────────────────────────────────────────────
# Recommended: 20.x LTS
# Check: node --version
